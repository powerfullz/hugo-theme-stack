{{/* prettier-ignore-start */}} /// Params: /// Resources: Where to search for images /// Image: Image URL /// Returns: /// Local: true if the image is local, false if it is remote /// Resource: Hugo image resource object, not nil if the image is local /// ColorResource: Hugo image resource object used for color extraction /// Permalink: Image URL /// Height: Image height /// Width: Image width {{/* prettier-ignore-end */}} {{ $result := "" }} {{ if .Image }} {{ $url := urls.Parse .Image }} {{ $resource := "" }} {{ if not (in (slice "https" "http") $url.Scheme) }} {{ $resource = .Resources.Get (printf "%s" (.Image | safeURL)) }} {{ else }} {{ with try (resources.GetRemote $url) }} {{ with .Err }} {{ warnf "Failed to get remote resource %q: %s" $url . }} {{ else with .Value }} {{ $resource = . }} {{ else }} {{ warnf "Unable to get remote resource %q" $url }} {{ end }} {{ end }} {{ end }} {{ if $resource }} {{- $isSVG := eq $resource.MediaType.SubType "svg" -}} {{- if $isSVG -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- else -}} {{- if (default true .Page.Site.Params.imageProcessing.content.enabled) -}} {{- $maxSize := default (default 2048 .Page.Site.Params.imageProcessing.content.maxWidth) .Page.Site.Params.imageProcessing.content.maxSize -}} {{- $processed := $resource -}} {{- $stripEXIF := default true .Page.Site.Params.imageProcessing.content.stripEXIF -}} {{- $quality := default 90 .Page.Site.Params.imageProcessing.content.quality -}} {{- $convertPNG := default false .Page.Site.Params.imageProcessing.content.convertPNG -}} {{- $sourceType := lower $processed.MediaType.SubType -}} {{- $shouldConvertToWebp := or (in (slice "jpeg" "jpg") $sourceType) (and (eq $sourceType "png") $convertPNG) -}} {{- $targetType := $sourceType -}} {{- if $shouldConvertToWebp -}} {{- $targetType = "webp" -}} {{- end -}} {{/* Build one Resize spec so resize/format conversion/EXIF stripping happen in a single re-encode. */}} {{- $resizeSpec := "" -}} {{- if and (gt $maxSize 0) (ge $processed.Width $processed.Height) (gt $processed.Width $maxSize) -}} {{- $resizeSpec = printf "%dx" $maxSize -}} {{- else if and (gt $maxSize 0) (lt $processed.Width $processed.Height) (gt $processed.Height $maxSize) -}} {{- $resizeSpec = printf "x%d" $maxSize -}} {{/* No resize needed, but keep one-pass re-encode for EXIF stripping or format conversion. */}} {{- else if or $stripEXIF (ne $targetType $sourceType) -}} {{- $resizeSpec = printf "%dx%d" $processed.Width $processed.Height -}} {{- end -}} {{- if $resizeSpec -}} {{/* Attach output format/quality directly to Resize options. */}} {{- $resizeOptions := $resizeSpec -}} {{- if eq $targetType "webp" -}} {{- $resizeOptions = printf "%s webp q%d" $resizeSpec $quality -}} {{- else if in (slice "jpeg" "jpg") $targetType -}} {{- $resizeOptions = printf "%s jpeg q%d" $resizeSpec $quality -}} {{- else if eq $targetType "png" -}} {{- $resizeOptions = printf "%s png" $resizeSpec -}} {{- else if eq $targetType "tiff" -}} {{- $resizeOptions = printf "%s tiff" $resizeSpec -}} {{- end -}} {{- with try ($processed.Resize $resizeOptions) -}} {{- with .Err -}} {{ warnf "Failed to process image %q: %s" $processed.RelPermalink . }} {{- $processed = $resource -}} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- end -}} {{- $result = dict "Local" true "Resource" $processed "ColorResource" $resource "Permalink" $processed.RelPermalink "Height" $processed.Height "Width" $processed.Width }} {{- else -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- else -}} {{ $result = dict "Local" false "Resource" nil "ColorResource" nil "Permalink" .Image "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- return $result -}}
