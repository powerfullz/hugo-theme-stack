{{/* prettier-ignore-start */}} /// Params: /// Resources: Where to search for images /// Image: Image URL /// Returns: /// Local: true if the image is local, false if it is remote /// Resource: Hugo image resource object, not nil if the image is local /// ColorResource: Hugo image resource object used for color extraction /// Permalink: Image URL /// Height: Image height /// Width: Image width {{/* prettier-ignore-end */}} {{ $result := "" }} {{ if .Image }} {{ $url := urls.Parse .Image }} {{ $resource := "" }} {{ if not (in (slice "https" "http") $url.Scheme) }} {{ $resource = .Resources.Get (printf "%s" (.Image | safeURL)) }} {{ else }} {{ with try (resources.GetRemote $url) }} {{ with .Err }} {{ warnf "Failed to get remote resource %q: %s" $url . }} {{ else with .Value }} {{ $resource = . }} {{ else }} {{ warnf "Unable to get remote resource %q" $url }} {{ end }} {{ end }} {{ end }} {{ if $resource }} {{- $isSVG := eq $resource.MediaType.SubType "svg" -}} {{- if $isSVG -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- else -}} {{- if (default true .Page.Site.Params.imageProcessing.content.enabled) -}} {{- $maxSize := default (default 2048 .Page.Site.Params.imageProcessing.content.maxWidth) .Page.Site.Params.imageProcessing.content.maxSize -}} {{- $processed := $resource -}} {{- if gt $maxSize 0 -}} {{- if and (ge $processed.Width $processed.Height) (gt $processed.Width $maxSize) -}} {{- with try ($processed.Resize (printf "%dx" $maxSize)) -}} {{- with .Err -}} {{ warnf "Failed to resize image %q: %s" $resource.RelPermalink . }} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- else if and (lt $processed.Width $processed.Height) (gt $processed.Height $maxSize) -}} {{- with try ($processed.Resize (printf "x%d" $maxSize)) -}} {{- with .Err -}} {{ warnf "Failed to resize image %q: %s" $resource.RelPermalink . }} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with try ($processed.Process "webp") -}} {{- with .Err -}} {{ warnf "Failed to process image as WebP %q: %s" $processed.RelPermalink . }} {{- $processed = $resource -}} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- $result = dict "Local" true "Resource" $processed "ColorResource" $resource "Permalink" $processed.RelPermalink "Height" $processed.Height "Width" $processed.Width }} {{- else -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- else -}} {{ $result = dict "Local" false "Resource" nil "ColorResource" nil "Permalink" .Image "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- return $result -}}
