{{/* prettier-ignore-start */}} /// Params: /// Resources: Where to search for images /// Image: Image URL /// Returns: /// Local: true if the image is local, false if it is remote /// Resource: Hugo image resource object, not nil if the image is local /// ColorResource: Hugo image resource object used for color extraction /// Permalink: Image URL /// Height: Image height /// Width: Image width {{/* prettier-ignore-end */}} {{ $result := "" }} {{ if .Image }} {{ $url := urls.Parse .Image }} {{ $resource := "" }} {{ $isRemote := in (slice "https" "http") $url.Scheme }} {{ if not $isRemote }} {{ $resource = .Resources.Get (printf "%s" (.Image | safeURL)) }} {{ end }} {{ if $resource }} {{- $isSVG := eq $resource.MediaType.SubType "svg" -}} {{- if $isSVG -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- else -}} {{- if (and (default true .Page.Site.Params.imageProcessing.content.enabled) (ne $resource.MediaType.SubType "gif")) -}} {{- $maxWidth := default 2048 .Page.Site.Params.imageProcessing.content.maxWidth -}} {{- $processed := $resource -}} {{- if gt $maxWidth 0 -}} {{- if gt $processed.Width $maxWidth -}} {{- with try ($processed.Resize (printf "%dx" $maxWidth)) -}} {{- with .Err -}} {{ warnf "Failed to resize image %q: %s" $resource.RelPermalink . }} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with try ($processed.Process "webp") -}} {{- with .Err -}} {{ warnf "Failed to process image as WebP %q: %s" $processed.RelPermalink . }} {{- $processed = $resource -}} {{- else with .Value -}} {{- $processed = . -}} {{- end -}} {{- end -}} {{- $result = dict "Local" true "Resource" $processed "ColorResource" $resource "Permalink" $processed.RelPermalink "Height" $processed.Height "Width" $processed.Width }} {{- else -}} {{ $result = dict "Local" true "Resource" $resource "ColorResource" $resource "Permalink" $resource.RelPermalink "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- else -}} {{ $result = dict "Local" false "Resource" nil "ColorResource" nil "Permalink" .Image "Height" nil "Width" nil }} {{- end -}} {{- end -}} {{- return $result -}}
