{{- /* Params: Resource: Hugo image resource object (Optional, for processing) Width: Image width (Required) Height: Image height (Required) Resize: Whether to perform resize (Optional, default: false) Process: Map of image processing options - Format: Output format (Optional) - Quality: Output quality (Optional) Attributes: Map of HTML attributes for the tag */ -}} {{- $resource := .Resource -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- $resize := .Resize -}} {{- $process := .Process | default dict -}} {{- $attributes := .Attributes | default dict -}} {{- $format := lower (default "" $process.Format) -}} {{- $quality := int (default 0 $process.Quality) -}} {{- $options := slice -}} {{- if $format -}} {{- $options = $options | append $format -}} {{- end -}} {{- if gt $quality 0 -}} {{- $options = $options | append (printf "q%d" $quality) -}} {{- end -}} {{- if and $resize $resource (reflect.IsImageResourceProcessable $resource) $width $height -}} {{/* Create thumbnail with 1x/2x descriptors */}} {{- $srcset := slice -}} {{- range (slice 1 2) -}} {{- $w := mul $width . -}} {{- $h := mul $height . -}} {{- if and (le $w $resource.Width) (le $h $resource.Height) -}} {{- $fillSpec := printf "%dx%d" $w $h -}} {{- if gt (len $options) 0 -}} {{- $fillSpec = printf "%s %s" $fillSpec (delimit $options " ") -}} {{- end -}} {{- $resized := $resource.Fill $fillSpec -}} {{- $srcset = $srcset | append (printf "%s %dx" $resized.RelPermalink .) -}} {{- if eq . 1 -}} {{- $attributes = merge $attributes (dict "src" $resized.RelPermalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $srcset) 0 -}} {{- $attributes = merge $attributes (dict "width" $width "height" $height "srcset" (delimit $srcset ", ")) -}} {{- end -}} {{- end -}}
